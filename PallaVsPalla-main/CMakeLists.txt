cmake_minimum_required(VERSION 3.16)
project(PallaVsPalla LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5)
FetchContent_MakeAvailable(raylib)
add_executable(PallaVsPalla src/main.cpp src/game.cpp src/ball_vm.cpp src/ai_client.cpp src/audio_synth.c)
target_link_libraries(PallaVsPalla PRIVATE raylib)
if (WIN32)
    target_link_libraries(PallaVsPalla PRIVATE winmm gdi32 opengl32 winhttp)
endif()
